function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('⚡ PERSEBA Web')
    .addItem('🚀 Setup / Perbarui Struktur Spreadsheet', 'setupDatabase')
    .addItem('📊 Cek Status & Info Database', 'tampilkanInfoDatabase')
    .addItem('🧹 Buat Contoh Pesan Uji Coba', 'buatContohPesan')
    .addToUi();
}

function setupDatabase() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  try {
    ss.setSpreadsheetTimeZone("Asia/Jakarta");
  } catch (e) {
    Logger.log("Timezone set: " + e.message);
  }

  // 1. SHEET: Pesan_Kontak
  var sheetKontak = ss.getSheetByName("Pesan_Kontak");
  if (!sheetKontak) {
    sheetKontak = ss.insertSheet("Pesan_Kontak");
  }

  if (sheetKontak.getLastRow() === 0) {
    var headersKontak = [
      ["Waktu (WIB)", "Nama Pengirim", "No. WhatsApp / Telp", "Isi Pesan", "Status Pesan", "Catatan Admin"]
    ];
    sheetKontak.getRange(1, 1, 1, 6).setValues(headersKontak);
    sheetKontak.getRange(1, 1, 1, 6)
      .setFontWeight("bold")
      .setBackground("#15803d")
      .setFontColor("#ffffff")
      .setHorizontalAlignment("center")
      .setVerticalAlignment("middle");
    
    sheetKontak.setRowHeight(1, 38);
    sheetKontak.setFrozenRows(1);
    
    sheetKontak.setColumnWidth(1, 170);
    sheetKontak.setColumnWidth(2, 190);
    sheetKontak.setColumnWidth(3, 160);
    sheetKontak.setColumnWidth(4, 380);
    sheetKontak.setColumnWidth(5, 130);
    sheetKontak.setColumnWidth(6, 200);

    var ruleStatus = SpreadsheetApp.newDataValidation()
      .requireValueInList(["Baru", "Dibaca", "Dibalas", "Selesai"], true)
      .setAllowInvalid(false)
      .build();
    sheetKontak.getRange("E2:E500").setDataValidation(ruleStatus);
  }

  // 2. SHEET: Data_Website
  var sheetData = ss.getSheetByName("Data_Website");
  if (!sheetData) {
    sheetData = ss.insertSheet("Data_Website");
  }

  if (sheetData.getLastRow() === 0) {
    var headersData = [
      ["KONTEN_JSON_LENGKAP", "TERAKHIR_DIPERBARUI", "KETERANGAN_STATUS"]
    ];
    sheetData.getRange(1, 1, 1, 3).setValues(headersData);
    sheetData.getRange(1, 1, 1, 3)
      .setFontWeight("bold")
      .setBackground("#d97706")
      .setFontColor("#ffffff")
      .setHorizontalAlignment("center")
      .setVerticalAlignment("middle");

    sheetData.setRowHeight(1, 38);
    sheetData.setFrozenRows(1);
    sheetData.setColumnWidth(1, 550);
    sheetData.setColumnWidth(2, 200);
    sheetData.setColumnWidth(3, 250);

    sheetData.getRange("A2").setValue('{"keterangan":"Siap menerima sinkronisasi dari website PERSEBA"}');
    sheetData.getRange("B2").setValue(Utilities.formatDate(new Date(), "Asia/Jakarta", "dd-MM-yyyy HH:mm:ss"));
    sheetData.getRange("C2").setValue("Menunggu sinkronisasi pertama");
  }

  // 3. SHEET: Pengaturan
  var sheetSetting = ss.getSheetByName("Pengaturan");
  if (!sheetSetting) {
    sheetSetting = ss.insertSheet("Pengaturan");
  }

  if (sheetSetting.getLastRow() === 0) {
    var headersSetting = [
      ["Parameter", "Nilai", "Keterangan"]
    ];
    sheetSetting.getRange(1, 1, 1, 3).setValues(headersSetting);
    sheetSetting.getRange(1, 1, 1, 3)
      .setFontWeight("bold")
      .setBackground("#1e293b")
      .setFontColor("#ffffff")
      .setHorizontalAlignment("center");

    sheetSetting.setRowHeight(1, 34);
    sheetSetting.setFrozenRows(1);
    sheetSetting.setColumnWidth(1, 240);
    sheetSetting.setColumnWidth(2, 300);
    sheetSetting.setColumnWidth(3, 300);

    var settingsData = [
      ["NAMA_ORGANISASI", "PERSEBA (Persatuan Remaja Barat Sawah)", "Identitas resmi organisasi"],
      ["DESA_DUSUN", "Dusun Baratsawah, Desa Gunungteguh, Bawean", "Lokasi sekretariat"],
      ["NOTIFIKASI_EMAIL_AKTIF", "FALSE", "Ubah jadi TRUE jika ingin email saat pesan baru masuk"],
      ["EMAIL_ADMIN", Session.getActiveUser().getEmail() || "admin@baratsawah.id", "Alamat email penerima notifikasi"],
      ["NOMOR_WA_ADMIN", "6281234567890", "Nomor WhatsApp pengurus untuk tindak lanjut pesan"]
    ];
    sheetSetting.getRange(2, 1, settingsData.length, 3).setValues(settingsData);
  }

  try {
    var defaultSheet = ss.getSheetByName("Sheet1");
    if (defaultSheet && ss.getSheets().length > 1 && defaultSheet.getLastRow() === 0) {
      ss.deleteSheet(defaultSheet);
    }
  } catch (err) {}

  try {
    var ui = SpreadsheetApp.getUi();
    ui.alert(
      "✅ Inisialisasi Berhasil!",
      "Struktur database PERSEBA Baratsawah telah siap:\n\n" +
      "1. Sheet 'Pesan_Kontak' (Formulir Masuk)\n" +
      "2. Sheet 'Data_Website' (Sinkronisasi Konten)\n" +
      "3. Sheet 'Pengaturan' (Konfigurasi Admin)\n\n" +
      "Langkah selanjutnya:\n" +
      "Klik tombol 'Deploy' > 'New deployment' > Web app untuk mendapatkan Web App URL.",
      ui.ButtonSet.OK
    );
  } catch (e) {
    Logger.log("Inisialisasi selesai via script.");
  }

  return { status: "success", message: "Struktur database PERSEBA siap digunakan" };
}

function tampilkanInfoDatabase() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetKontak = ss.getSheetByName("Pesan_Kontak");

    "📊 Status Database PERSEBA",

    "• Total Pesan Masuk: " + totalPesan + " pesan\n" +
    "• Terakhir Sinkron Website: " + lastUpdate + "\n" +
    "• ID Spreadsheet: " + ss.getId(),
    ui.ButtonSet.OK
  );
}

/**
 * ==========================================================================
 * PERSEBA DUSUN BARATSAWAH - GOOGLE APPS SCRIPT
 * File: Code.gs
 * Fungsi: Controller Web App, REST API (doGet / doPost), & Jembatan Database
 * ==========================================================================
 */

function doGet(e) {
  try {
    e = e || { parameter: {} };
    var action = e.parameter.action || "";

    if (action === "get_data" || action === "download_backup" || e.parameter.api === "true") {
      return kirimDataWebsiteJson();
    }

    if (action === "get_contacts") {
      return kirimDataPesanKontakJson();
    }

    if (action === "ping" || action === "test") {
      return jsonResponse({
        status: "success",
        message: "⚡ Google Apps Script PERSEBA aktif dan terhubung!",
        timestamp: new Date().toISOString(),
        spreadsheet: SpreadsheetApp.getActiveSpreadsheet().getName()
      });
    }

    // Tampilkan Web App Index.html jika diakses di peramban
    var template = HtmlService.createTemplateFromFile("Index");
    var html = template.evaluate();
    html.setTitle("Portal & Database PERSEBA Baratsawah");
    html.addMetaTag("viewport", "width=device-width, initial-scale=1");
    html.setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    return html;

  } catch (err) {
    return jsonResponse({
      status: "error",
      message: "Terjadi kesalahan di server Apps Script: " + err.toString()
    });
  }
}

function doPost(e) {
  try {
    var raw = e && e.postData ? e.postData.contents : null;
    var data = {};

    if (raw) {
      try {
        data = JSON.parse(raw);
      } catch (ex) {
        data = (e && e.parameter) ? e.parameter : {};
      }
    } else {
      data = (e && e.parameter) ? e.parameter : {};
    }

    var action = data.action || "contact_form";
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Simpan Formulir Kontak
    if (action === "contact_form") {
      var sheetKontak = ss.getSheetByName("Pesan_Kontak");
      if (!sheetKontak) {
        setupDatabase();
        sheetKontak = ss.getSheetByName("Pesan_Kontak");
      }

      var waktu = Utilities.formatDate(new Date(), "Asia/Jakarta", "dd-MM-yyyy HH:mm:ss");
      var nama = (data.nama || "-").toString().trim();
      var telp = (data.telp || "-").toString().trim();
      var pesan = (data.pesan || "-").toString().trim();

      sheetKontak.appendRow([
        waktu,
        nama,
        "'" + telp,
        pesan,
        "Baru",
        "Diterima via formulir website"
      ]);

      try {
        kirimNotifikasiEmailAdmin(nama, telp, pesan, waktu);
      } catch (mailErr) {}

      return jsonResponse({
        status: "success",
        message: "Pesan kontak berhasil dicatat di Google Sheets PERSEBA",
        timestamp: waktu
      });
    }

    // 2. Simpan Seluruh Data Website (JSON Backup)
    if (action === "sync_website_data") {
      var sheetData = ss.getSheetByName("Data_Website");
      if (!sheetData) {
        setupDatabase();
        sheetData = ss.getSheetByName("Data_Website");
      }

      var payloadString = typeof data.payload === "string" ? data.payload : JSON.stringify(data.payload || {});
      var updateTime = Utilities.formatDate(new Date(), "Asia/Jakarta", "dd-MM-yyyy HH:mm:ss");

      sheetData.getRange("A2").setValue(payloadString);
      sheetData.getRange("B2").setValue(updateTime);
      sheetData.getRange("C2").setValue("Sinkronisasi otomatis berhasil");

      return jsonResponse({
        status: "success",
        message: "Data website berhasil disinkronkan ke Google Sheets pada " + updateTime,
        timestamp: updateTime
      });
    }

    // 3. Uji Koneksi POST
    if (action === "test_connection") {
      return jsonResponse({
        status: "success",
        message: "Uji koneksi POST berhasil! Google Apps Script siap menerima data.",
        timestamp: new Date().toISOString()
      });
    }

    return jsonResponse({
      status: "error",
      message: "Aksi tidak dikenali"
    });

  } catch (err) {
    return jsonResponse({
      status: "error",
      message: "Gagal memproses POST: " + err.toString()
    });
  }
}

function jsonResponse(obj) {

function kirimDataWebsiteJson() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetData = ss.getSheetByName("Data_Website");

  if (!sheetData || sheetData.getLastRow() < 2) {
    return jsonResponse({
      status: "success",
      message: "Sheet Data_Website belum memiliki data tersimpan",
      data: null
    });
  }

  var jsonVal = sheetData.getRange("A2").getValue();
  var lastUpdate = sheetData.getRange("B2").getValue();
  var parsed = null;

  try {
    parsed = typeof jsonVal === "string" ? JSON.parse(jsonVal) : jsonVal;
  } catch (e) {
    parsed = jsonVal;
  }

  return jsonResponse({
    status: "success",
    timestamp: lastUpdate,
    data: parsed
  });
}

function kirimDataPesanKontakJson() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Pesan_Kontak");

  if (!sheet || sheet.getLastRow() < 2) {
    return jsonResponse({ status: "success", total: 0, data: [] });

  var rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues();
  var list = [];

  for (var i = rows.length - 1; i >= 0; i--) {

      row: i + 2,

      nama: rows[i][1],
      telp: rows[i][2],
      pesan: rows[i][3],
      status: rows[i][4],
      catatan: rows[i][5]
    });
  }

  return jsonResponse({ status: "success", total: list.length, data: list });
}

function kirimNotifikasiEmailAdmin(nama, telp, pesan, waktu) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetSetting = ss.getSheetByName("Pengaturan");
  if (!sheetSetting) return;

  var settingRows = sheetSetting.getDataRange().getValues();
  var aktif = false;
  var emailTujuan = "";

  for (var i = 1; i < settingRows.length; i++) {
    if (settingRows[i][0] === "NOTIFIKASI_EMAIL_AKTIF" && String(settingRows[i][1]).toUpperCase() === "TRUE") {
      aktif = true;
    }
    if (settingRows[i][0] === "EMAIL_ADMIN") {
      emailTujuan = String(settingRows[i][1]).trim();
    }
  }

  if (aktif && emailTujuan && emailTujuan.indexOf("@") !== -1) {
    var subject = "📬 Pesan Baru dari Website PERSEBA: " + nama;
    var body = "Ada pesan masuk baru di website PERSEBA Dusun Baratsawah:\n\n" +
               "Waktu: " + waktu + "\n" +
               "Pengirim: " + nama + "\n" +
               "No. Telp / WA: " + telp + "\n" +
               "Isi Pesan:\n" + pesan + "\n\n" +
               "---\n" +
               "Buka Google Spreadsheet untuk mengelola: " + ss.getUrl();
    
    MailApp.sendEmail(emailTujuan, subject, body);
  }
}

function getWebAppData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetKontak = ss.getSheetByName("Pesan_Kontak");
  var sheetData = ss.getSheetByName("Data_Website");

  var totalPesan = 0;
  var pesanBaru = 0;
  var pesanList = [];

  if (sheetKontak && sheetKontak.getLastRow() >= 2) {
    var dataKontak = sheetKontak.getRange(2, 1, sheetKontak.getLastRow() - 1, 6).getValues();
    totalPesan = dataKontak.length;

    for (var i = dataKontak.length - 1; i >= 0; i--) {
      var st = String(dataKontak[i][4] || "Baru").trim();
      if (st.toLowerCase() === "baru") {
        pesanBaru++;
      }
      if (pesanList.length < 25) {
        pesanList.push({
          row: i + 2,
          waktu: dataKontak[i][0],
          nama: dataKontak[i][1],
          telp: dataKontak[i][2],
          pesan: dataKontak[i][3],
          status: st,
          catatan: dataKontak[i][5] || ""
        });
      }
    }
  }

  var lastSync = "Belum pernah sinkron";
  if (sheetData && sheetData.getLastRow() >= 2) {
    lastSync = sheetData.getRange("B2").getValue() || lastSync;
  }

  return {
    spreadsheetName: ss.getName(),
    spreadsheetUrl: ss.getUrl(),
    totalPesan: totalPesan,
    pesanBaru: pesanBaru,
    lastSync: lastSync,
    pesanList: pesanList
  };
}

function updateStatusPesanDariWeb(row, newStatus) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Pesan_Kontak");
  if (!sheet) return { success: false, message: "Sheet Pesan_Kontak tidak ditemukan" };

  row = parseInt(row, 10);
  if (row >= 2 && row <= sheet.getLastRow()) {
    sheet.getRange(row, 5).setValue(newStatus);
    return { success: true, message: "Status baris " + row + " diubah menjadi " + newStatus };
  }
  return { success: false, message: "Nomor baris tidak valid" };
}
